// Devuelve la fecha y hora actual del RTC en una cadena
String getDateTimeString() {
  if (!rtc.isrunning()) {
    return "RTC no ajustado";
  }
  DateTime now = rtc.now();
  char buf[25];
  snprintf(buf, sizeof(buf), "%02d/%02d/%04d %02d:%02d:%02d",
           now.day(), now.month(), now.year(),
           now.hour(), now.minute(), now.second());
  return String(buf);
}

  float t = dht.readTemperature();
  float h = dht.readHumidity();
  if (isnan(t) || isnan(h)) {
    t = lastData.temp;
    h = lastData.hum;
  }

  // MQ-135
  int rawMQ = analogRead(MQ135_PIN);
  float ppm = map(rawMQ, 0, 4095, (int)MQ_PPM_MIN, (int)MQ_PPM_MAX);
  String mqState = classifyAirQuality(ppm);

  // Luz
  int rawL = analogRead(LDR_PIN);
  float vL = (rawL * 3.3f) / 4095.0f;
  bool isNight = vL > 1.6;           // umbral aproximado
  String mode = isNight ? "Noche" : "Dia";

  // Fecha/hora
  String dt = getDateTimeString();

  // Actualización de la estructura con las últimas mediciones
  lastData.temp      = t;
  lastData.hum       = h;
  lastData.mqPPM     = ppm;
  lastData.mqState   = mqState;
  lastData.lightRaw  = rawL;
  lastData.lightVolt = vL;
  lastData.lightMode = mode;
  lastData.isNight   = isNight;
  lastData.datetime  = dt;
}

// Actualiza el contenido mostrado en la pantalla OLED con los últimos datos
void updateOLED() {
  oled.clearBuffer();
  oled.setFont(u8g2_font_6x12_tf);

  // Línea 1: Temp y Humedad
  String l1 = "T:" + String(lastData.temp, 1) + "C  H:" + String(lastData.hum, 0) + "%";
  oled.drawStr(0, 12, l1.c_str());

  // Línea 2: Cal + PPM (abreviado)
  String l2 = "Cal:" + lastData.mqState + " PPM:" + String((int)lastData.mqPPM);
  oled.drawStr(0, 26, l2.c_str());

  // Línea 3: Modo Día/Noche
  String l3 = "Modo:" + lastData.lightMode;
  oled.drawStr(0, 40, l3.c_str());

  // Línea 4: Fecha y hora
  oled.drawStr(0, 54, lastData.datetime.c_str());

  // Línea 5: Estado del sistema
  oled.setFont(u8g2_font_6x10_tf);
  if (systemPaused) {
    oled.drawStr(0, 64, "Sistema Detenido");
  } else {
    oled.drawStr(0, 64, "Sistema Activo");
  }

  oled.sendBuffer();
}

// ===================== HANDLERS HTTP ================================

void handleRoot() {
  server.send_P(200, "text/html", MAIN_page);
}

// Respuesta JSON con las últimas lecturas de todos los sensores
void handleSensorJson() {
  String json = "{";
  json += "\"temp\":"      + String(lastData.temp, 2)      + ",";
  json += "\"hum\":"       + String(lastData.hum, 2)       + ",";
  json += "\"mqPPM\":"     + String(lastData.mqPPM, 0)     + ",";
  json += "\"mqState\":\"" + lastData.mqState              + "\",";
  json += "\"lightRaw\":"  + String(lastData.lightRaw)     + ",";
  json += "\"lightVolt\":" + String(lastData.lightVolt, 3) + ",";
  json += "\"mode\":\""    + lastData.lightMode            + "\",";
  json += "\"isNight\":"   + String(lastData.isNight ? "true" : "false") + ",";
  json += "\"datetime\":\""+ lastData.datetime + "\"";
  json += "}";
  server.send(200, "application/json", json);
}
 
// Handler que marca el sistema como pausado 
void handlePause() {
  systemPaused = true;
  server.send(200, "text/plain", "paused");
}

// Handler que reanuda el sistema
void handleResume() {
  systemPaused = false;
  server.send(200, "text/plain", "resumed");
}

// ===================== SETUP / LOOP ================================

void setup() {
  Serial.begin(115200);
  delay(200);

  dht.begin();   // Inicialización del sensor DHT11
  Wire.begin();  // Inicio de bus I2C para RTC y posibles periféricos I2C

if (!rtc.begin()) {
  Serial.println("No se encuentra el RTC");
} else {
  Serial.println("RTC encontrado, ajustando a fecha/hora de compilación...");
  rtc.adjust(DateTime(F(__DATE__), F(__TIME__)));    // Ajuste inicial del RTC
}

 // Inicialización de la pantalla OLED
  oled.begin();
  oled.clearBuffer();
  oled.setFont(u8g2_font_6x12_tf);
  oled.drawStr(0, 16, "Iniciando sistema...");
  oled.sendBuffer();

 // Conexión a la red WiFi
  WiFi.begin(ssid, password);
  Serial.print("Conectando a WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println();
  Serial.print("Conectado. IP: ");
  Serial.println(WiFi.localIP());

// Registro de rutas HTTP para la interfaz web
  server.on("/",       HTTP_GET, handleRoot);
  server.on("/sensor", HTTP_GET, handleSensorJson);
  server.on("/pause",  HTTP_GET, handlePause);
  server.on("/resume", HTTP_GET, handleResume);
  server.begin();

 // Primera lectura de sensores y primera actualización de la pantalla
  readSensors();
  updateOLED();
}

void loop() {
  server.handleClient();

  unsigned long now = millis();

  // Sensores solo si el sistema NO está pausado
  if (!systemPaused && now - lastSensorMillis >= SENSOR_INTERVAL_MS) {
    lastSensorMillis = now;
    readSensors();
  }

  // OLED siempre se actualiza 
  if (now - lastOledMillis >= OLED_INTERVAL_MS) {
    lastOledMillis = now;
    updateOLED();
  }
}
