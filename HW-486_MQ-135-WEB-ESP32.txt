// -------- Función para clasificar el aire --------
// Recibe el valor ADC del MQ-135 y devuelve una etiqueta de calidad del aire
String calidad(int raw) {
  if (raw < 900) return "Excelente";   // Muy baja concentración de gases
  if (raw < 1500) return "Buena";      // Aún en rango bueno
  if (raw < 2200) return "Aceptable";  // Algo elevado
  if (raw < 3000) return "Mala";       // Alta concentración
  return "Muy mala";                   // Muy alta concentración
}

// -------- Ruta /sensor --------
// Esta función se ejecuta cuando el navegador pide la URL
// Se lee los sensores,  se arma un JSON y se envía como respuesta
void handleSensor() {
  // --- Lectura MQ-135 (calidad de aire) ---
  int mqRaw = analogRead(MQ135_PIN);              // Lee el dato crudo del ADC 
  float mqVolt = mqRaw * (3.3 / 4095.0);          // Convierte el valor ADC a voltaje real (3.3V de referencia)
  String state = calidad(mqRaw);                  // Determina la calidad del aire en texto usando la función anterior

  // --- Lectura sensor de Luz (HW-486) ---
  int lightRaw = analogRead(LIGHT_PIN);           // Lee el valor analógico del sensor de luz
  float lightVolt = lightRaw * (3.3 / 4095.0);    // Convierte ese valor a voltaje

  // Define si es "día" o "noche" según el nivel de luz medido
 
  bool isNight = (lightRaw > 2000);               // Umbral definido
  String mode = isNight ? "Noche" : "Día";        // Modo en texto para mostrar en la página

  // --- Construcción manual del JSON que se enviará al navegador ---
  String json = "{";
  json += "\"raw\":" + String(mqRaw) + ",";               // Valor ADC del MQ-135
  json += "\"volt\":" + String(mqVolt, 3) + ",";          // Voltaje del MQ-135
  json += "\"state\":\"" + state + "\",";                 // Estado de la calidad del aire
  json += "\"lightRaw\":" + String(lightRaw) + ",";       // Valor ADC del sensor de luz
  json += "\"lightVolt\":" + String(lightVolt, 3) + ",";  // Voltaje del sensor de luz
  json += "\"isNight\":" + String(isNight ? 1 : 0) + ","; // 1 si es noche, 0 si es día
  json += "\"mode\":\"" + mode + "\"";                    // Texto "Día" o "Noche"
  json += "}";

  // Envía la respuesta HTTP con código 200 y contenido JSON
  server.send(200, "application/json", json);
}

// -------- Setup --------
void setup() {
  Serial.begin(115200);          
  pinMode(MQ135_PIN, INPUT);     // Configura el pin del MQ-135 como entrada analógica
  pinMode(LIGHT_PIN, INPUT);     // Configura el pin del sensor de luz como entrada analógica

  // --- Conexión a la red WiFi ---
  WiFi.begin(ssid, password);    
  Serial.println("Conectando a WiFi...");
  while (WiFi.status() != WL_CONNECTED) { 
    delay(500);
    Serial.print(".");           
  }
  Serial.println("\nConectado!");
  Serial.print("IP: ");
  Serial.println(WiFi.localIP()); 

  // --- Rutas del servidor web ---
  server.on("/", []() {
    server.send_P(200, "text/html", MAIN_page);
  });

  // llama a handleSensor para devolver datos en JSON
  server.on("/sensor", handleSensor);

  
  server.begin();
  Serial.println("Servidor iniciado.");
}

// -------- Loop --------
void loop() {
  server.handleClient();
}
